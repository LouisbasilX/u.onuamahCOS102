        "# Original diabetes data with missing values (for simulation)\n",
        "df_raw = diabetes_df.copy()\n",
        "columns_with_na = ['Glucose', 'BloodPressure', 'SkinThickness', 'Insulin', 'BMI']\n",
        "\n",
        "col_dropdown = widgets.Dropdown(options=columns_with_na, description='Column:')\n",
        "method_dropdown = widgets.Dropdown(options=['mean', 'median'], description='Method:')\n",
        "\n",
        "def impute_and_show(col, method):\n",
        "    df_sim = df_raw.copy()\n",
        "    if method == 'mean':\n",
        "        df_sim[col] = df_sim[col].fillna(df_sim[col].mean())\n",
        "    else:\n",
        "        df_sim[col] = df_sim[col].fillna(df_sim[col].median())\n",
        "    \n",
        "    print(f\"Missing values after imputation: {df_sim[col].isna().sum()}\")\n",
        "    print(\"\\nDescriptive statistics of the column:\")\n",
        "    display(df_sim[col].describe())\n",
        "    \n",
        "    # Show before/after comparison\n",
        "    plt.figure(figsize=(8,4))\n",
        "    plt.subplot(1,2,1)\n",
        "    df_raw[col].dropna().hist(bins=20, color='#e94560', alpha=0.7, label='Original (non‑null)')\n",
        "    plt.title('Original Distribution')\n",
        "    plt.xlabel(col)\n",
        "    plt.subplot(1,2,2)\n",
        "    df_sim[col].hist(bins=20, color='#64ffda', alpha=0.7)\n",

        "    plt.xlabel(col)\n",
        "    plt.tight_layout()\n",
        "    plt.show()\n",
        "\n",
